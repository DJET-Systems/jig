Manifest reference and sample sheets

The contract for jig.toml, jig.lock and config.toml. Unknown keys are errors (DJ1003). A JSON Schema is generated from the parser by a test (T-003-03) into jig.schema.json next to this file; if this sheet and the schema disagree, CI fails.

Sample sheet

S1. C++23 executable (what jig init writes)

[project]
name     = "hello"
version  = "0.1.0"
kind     = "exe"
language = "c++"
std      = "c++23"

[toolchain]
vendor  = "llvm"
version = "20"

[test]
sources = ["tests/**/*.cpp"]

S2. C17 static library, strict warnings, sanitizers in debug

[project]
name     = "ringbuf"
version  = "1.2.0"
license  = "GPL-3.0-only"
kind     = "static-lib"
language = "c"
std      = "c17"

[toolchain]
vendor      = "llvm"
version     = "^19"
hermeticity = "h1"

[build]
warnings       = "strict"
public-include = ["include"]

[profile.debug]
sanitizers = ["address", "undefined"]

[profile.release]
lto = "thin"

S3. Header-only C++20 library

[project]
name     = "smallvec"
kind     = "header-only"
language = "c++"
std      = "c++20"

[toolchain]
version = ">=17,<21"

S4. Existing CMake project, adopted

[project]
name     = "legacy-app"
kind     = "exe"
language = "c++"
std      = "c++17"

[toolchain]
vendor  = "llvm"
version = "18.1"

[build]
backend    = "cmake"
cmake-args = ["-DLEGACY_FEATURE=ON"]

S5. Use the machine's compiler (H0)

[project]
name     = "quick"
kind     = "exe"
language = "c"
std      = "c11"

[toolchain]
vendor      = "system"
hermeticity = "h0"

[toolchain.system]
cc = "gcc"

S6. Several targets in one project

[project]
name     = "suite"
kind     = "exe"
language = "c++"
std      = "c++23"

[[target]]
name    = "suite-core"
kind    = "static-lib"
sources = ["src/core/**/*.cpp"]

[[target]]
name         = "suite-cli"
kind         = "exe"
sources      = ["src/cli/**/*.cpp"]
link-targets = ["suite-core"]

S7. Environment control

[env]
inherit = "none"
set     = { SUITE_MODE = "ci", TZ = "UTC" }
